[Fact]
public async Task RotateSecret_Concurrent_ShouldBeThreadSafe()
{
    // Arrange: Create subscription
    var subscriptionId = await CreateTestSubscriptionAsync();
    
    // Act: Attempt concurrent secret rotations
    var tasks = Enumerable.Range(0, 5).Select(_ => 
        _client.PostAsync($"/api/v1/webhooks/subscriptions/{subscriptionId}/rotate-secret", 
            new StringContent("\"new-secret\"", Encoding.UTF8, "application/json")))
        .ToArray();
    
    var responses = await Task.WhenAll(tasks);
    
    // Assert: Only one should succeed, others should fail gracefully
    responses.Count(r => r.IsSuccessStatusCode).Should().Be(1);

}
