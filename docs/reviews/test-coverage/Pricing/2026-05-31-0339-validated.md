[Collection("Pricing Integration")]
public class PriceRuleRepositoryTests : IAsyncLifetime
{
    private readonly PricingTestFactory _factory;
    private IPriceRuleRepository _repository = null!;

    [Fact]
    public async Task AddAsync_ConcurrentRuleCreation_BothSucceed()
    {
        // Arrange
        var productId = Guid.NewGuid();
        var rule1 = PriceRule.Create(productId, null, 1, DiscountType.Percentage, 10m);
        var rule2 = PriceRule.Create(productId, null, 2, DiscountType.FixedAmount, 5m);
        
        // Act
        var tasks = new[]
        {
            _repository.AddAsync(rule1),
            _repository.AddAsync(rule2)
        };
        await Task.WhenAll(tasks);
        await _repository.SaveChangesAsync();
        
        // Assert  
        var rules = await _repository.GetByProductIdAsync(productId);
        rules.Should().HaveCount(2);
    }
}
