```csharp
[Collection("Orders Integration")]
public class DemoIdempotencyControllerTests(OrdersWebAppFactory factory) : IAsyncLifetime
{
    private readonly HttpClient _client = factory.CreateClient();

    public Task InitializeAsync() => factory.EnsureSchemaAsync();
    public Task DisposeAsync() => Task.CompletedTask;

    [Fact]
    public async Task Claim_WithValidKey_ReturnsClaim()
    {
        var response = await _client.PostAsync("/demo/idempotency/claim", null, new Dictionary<string, string>
        {
            ["X-Idempotency-Key"] = "test-key-" + Guid.NewGuid()
        });

        response.Should().BeSuccessful();
        var claim = await response.Content.ReadFromJsonAsync<IdempotencyClaim>();
        claim.Should().NotBeNull();
        claim!.IsWinner.Should().BeTrue();
        claim.IsDuplicate.Should().BeFalse();
    }

    [Fact] 
    public async Task Claim_WithDuplicateKey_ReturnsDuplicate()
    {
        var key = "duplicate-test-" + Guid.NewGuid();
        
        // First claim
        await _client.PostAsync("/demo/idempotency/claim", null, 
            new Dictionary<string, string> { ["X-Idempotency-Key"] = key });

        // Second claim (duplicate)  
        var response = await _client.PostAsync("/demo/idempotency/claim", null,
            new Dictionary<string, string> { ["X-Idempotency-Key"] = key });

        var claim = await response.Content.ReadFromJsonAsync<IdempotencyClaim>();
        claim!.IsWinner.Should().BeFalse();
        claim.IsDuplicate.Should().BeTrue();
    }

    [Fact]
    public async Task Race_WithConcurrentRequests_HasSingleWinner()
    {
        var raceKey = "race-test-" + Guid.NewGuid();
        var request = new { Key = raceKey, Count = 5, TtlSeconds = 30 };

        var response = await _client.PostAsJsonAsync("/demo/idempotency/race", request);

        response.Should().BeSuccessful();
        var result = await response.Content.ReadFromJsonAsync<RaceResult>();
        result!.Outcomes.Should().HaveCount(5);
        result.Outcomes.Count(o => o.IsWinner).Should().Be(1, "exactly one winner in race");
    }

    private record IdempotencyClaim(string IdempotencyKey, Guid ClaimId, bool IsWinner, bool IsDuplicate);
    private record RaceResult(string Key, int Count, IReadOnlyList<RaceOutcome> Outcomes);
    private record RaceOutcome(int RequestIndex, bool IsWinner, Guid OrderId, long DurationMs);
}
```
