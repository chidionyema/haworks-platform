```csharp
public sealed class ApproveMerchantCommandHandlerTests
{
    [Fact]
    public async Task Handle_ValidPendingMerchant_ActivatesAndPublishesEvent()
    {
        // Arrange
        var context = CreateInMemoryContext();
        var publishEndpoint = new Mock<IPublishEndpoint>();
        var merchant = MerchantProfile.Create(Guid.NewGuid(), "Test", "test");
        context.Merchants.Add(merchant);
        await context.SaveChangesAsync();
        
        var handler = new ApproveMerchantCommandHandler(context, publishEndpoint.Object);
        var command = new ApproveMerchantCommand(merchant.Id, "admin-123");

        // Act
        var result = await handler.Handle(command, CancellationToken.None);

        // Assert
        result.IsSuccess.Should().BeTrue();
        merchant.Status.Should().Be(MerchantStatus.Active);
        publishEndpoint.Verify(p => p.Publish(
            It.Is<MerchantActivatedEvent>(e => e.MerchantId == merchant.Id), 
            It.IsAny<CancellationToken>()), Times.Once);
    }

    [Fact] 
    public async Task Handle_NonexistentMerchant_ReturnsNotFound()
    {
        // Arrange
        var context = CreateInMemoryContext();
        var publishEndpoint = new Mock<IPublishEndpoint>();
        var handler = new ApproveMerchantCommandHandler(context, publishEndpoint.Object);
        var command = new ApproveMerchantCommand(Guid.NewGuid(), "admin-123");

        // Act
        var result = await handler.Handle(command, CancellationToken.None);

        // Assert
        result.IsFailure.Should().BeTrue();
        result.Error.Code.Should().Be("Merchant.NotFound");
    }
}
```
