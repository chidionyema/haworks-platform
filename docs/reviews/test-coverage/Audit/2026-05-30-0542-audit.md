```csharp
public async Task WriteBatch_ConcurrentWrites_ShouldNotCorruptData()
{
    // Arrange
    var tasks = new List<Task>();
    var rows = Enumerable.Range(0, 100).Select(i => CreateTestAuditRow($"event-{i}")).ToList();
    
    // Act - simulate concurrent writes from multiple consumers
    for (int i = 0; i < 10; i++)
    {
        var batch = rows.Skip(i * 10).Take(10).ToList();
        tasks.Add(Task.Run(async () =>
        {
            foreach (var row in batch)
                await _auditWriter.WriteAsync(row, CancellationToken.None);
        }));
    }
    
    await Task.WhenAll(tasks);
    await _auditWriter.FlushAsync(CancellationToken.None);
    
    // Assert - all events should be persisted exactly once
    var count = await _dbContext.AuditEvents.CountAsync();
    count.Should().Be(100);
}
```
