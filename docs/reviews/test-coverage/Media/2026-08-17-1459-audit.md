## Missing Tests (ordered by priority)

#### HIGH: FileSignatureValidator has no coverage on a security control
- **For**: `Infrastructure/FileSignatureValidator.cs:ValidateAsync`
- **Type**: Unit
- **Why**: This is the magic-byte gate that blocks executables (MZ header) and MIME-mismatched uploads before virus scan. A regression here (e.g. broken webp sub-check at bytes 8-11, or truncated-stream handling) silently reopens the executable-upload vector with no test to catch it.
```csharp
public class FileSignatureValidatorTests
{
    private readonly FileSignatureValidator _sut = new(NullLogger<FileSignatureValidator>.Instance);

    [Fact]
    public async Task ValidateAsync_PngHeader_ReturnsValidPng()
    {
        var bytes = new byte[] { 0x89, 0x50, 0x4E, 0x47, 0x0D, 0x0A, 0x1A, 0x0A, 0,0,0,0 };
        using var stream = new MemoryStream(bytes);

        var result = await _sut.ValidateAsync(stream);

        result.IsValid.Should().BeTrue();
        result.FileType.Should().Be("image/png");
    }

    [Fact]
    public async Task ValidateAsync_MzHeader_ReturnsInvalid()
    {
        var bytes = new byte[] { 0x4D, 0x5A, 0x90, 0x00, 0,0,0,0,0,0,0,0 };
        using var stream = new MemoryStream(bytes);

        var result = await _sut.ValidateAsync(stream);

        result.IsValid.Should().BeFalse();
    }

    [Fact]
    public async Task ValidateAsync_WebpWithoutWebpMarker_ReturnsInvalid()
    {
        // RIFF header present but bytes 8-11 aren't "WEBP" — must not false-positive
        var bytes = new byte[] { 0x52, 0x49, 0x46, 0x46, 0,0,0,0, 0x41,0x56,0x49,0x20 };
        using var stream = new MemoryStream(bytes);

        var result = await _sut.ValidateAsync(stream);

        result.IsValid.Should().BeFalse();
    }

    [Fact]
    public async Task ValidateAsync_EmptyStream_ReturnsInvalid()
    {
        using var stream = new MemoryStream(Array.Empty<byte>());

        var result = await _sut.ValidateAsync(stream);

        result.IsValid.Should().BeFalse();
        result.FileType.Should().Be("Unknown");
    }
}
```

#### HIGH: BatchInitiateUploadHandler has zero coverage
- **For**: `Application/BatchInitiateUpload.cs:Handle`
- **Type**: Unit
- **Why**: Mixes dedup-by-hash, single-put, and multipart-init logic per item in one loop — no test verifies a batch with a duplicate-hash item alongside a new item doesn't cross-contaminate state, or that >50 files is rejected.
```csharp
public class BatchInitiateUploadTests
{
    private readonly MediaDbContext _context;
    private readonly Mock<IS3Service> _s3Mock = new();
    private readonly Mock<ICurrentUserService> _userMock = new();
    private readonly BatchInitiateUploadHandler _handler;

    public BatchInitiateUploadTests()
    {
        var options = new DbContextOptionsBuilder<MediaDbContext>()
            .UseInMemoryDatabase(Guid.NewGuid().ToString()).Options;
        _context = new MediaDbContext(options);
        _userMock.Setup(x => x.UserId).Returns("owner-1");
        _s3Mock.Setup(x => x.GeneratePreSignedUrl(It.IsAny<string>(), It.IsAny<string>())).Returns("http://url");
        _handler = new BatchInitiateUploadHandler(_context, _s3Mock.Object, _userMock.Object, Options.Create(new UploadOptions()));
    }

    [Fact]
    public async Task Handle_MixOfNewAndDuplicate_ReturnsCorrectFlagsPerFile()
    {
        var dupHash = new string('a', 64);
        var existing = MediaFile.Create("old.png", dupHash, 100, "image/png", "owner-1");
        _context.MediaFiles.Add(existing);
        await _context.SaveChangesAsync();

        var command = new BatchInitiateUploadCommand(new[]
        {
            new InitiateUploadCommand("new.png", dupHash, 100, "image/png"),
            new InitiateUploadCommand("fresh.png", new string('b', 64), 200, "image/png"),
        });

        var result = await _handler.Handle(command, CancellationToken.None);

        result.IsSuccess.Should().BeTrue();
        result.Value[0].AlreadyExists.Should().BeTrue();
        result.Value[0].Id.Should().Be(existing.Id);
        result.Value[1].AlreadyExists.Should().BeFalse();
    }
}
```
