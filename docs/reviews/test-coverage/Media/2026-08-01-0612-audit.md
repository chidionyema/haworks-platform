#### HIGH: FileSignatureValidator blocks/allows correctly by magic bytes
- **For**: `src/Media/Media.Api/Infrastructure/FileSignatureValidator.cs:38` (`ValidateAsync`)
- **Type**: Unit
- **Why**: This is the last line of defense against malware/executable uploads disguised with a fake `Content-Type`. Zero tests means a regression (e.g. broken WebP RIFF-subtype check, or MZ-header detection silently removed) ships undetected.
- **Skeleton**:
```csharp
using FluentAssertions;
using Haworks.Media.Api.Infrastructure;
using Microsoft.Extensions.Logging.Abstractions;
using Xunit;

namespace Haworks.Media.Unit;

public class FileSignatureValidatorTests
{
    private readonly FileSignatureValidator _validator = new(NullLogger<FileSignatureValidator>.Instance);

    [Fact]
    public async Task ValidateAsync_PngHeader_ReturnsValidPng()
    {
        var bytes = new byte[] { 0x89, 0x50, 0x4E, 0x47, 0x0D, 0x0A, 0x1A, 0x0A, 0,0,0,0 };
        await using var stream = new MemoryStream(bytes);

        var result = await _validator.ValidateAsync(stream);

        result.IsValid.Should().BeTrue();
        result.FileType.Should().Be("image/png");
    }

    [Fact]
    public async Task ValidateAsync_WebpMissingRiffWebpTag_ReturnsInvalid()
    {
        // RIFF header present but bytes 8-11 are not "WEBP" — must not match
        var bytes = new byte[] { 0x52, 0x49, 0x46, 0x46, 0,0,0,0, 0x00,0x00,0x00,0x00 };
        await using var stream = new MemoryStream(bytes);

        var result = await _validator.ValidateAsync(stream);

        result.IsValid.Should().BeFalse();
    }

    [Fact]
    public async Task ValidateAsync_ExecutableMzHeader_ReturnsInvalid()
    {
        var bytes = new byte[] { 0x4D, 0x5A, 0x90, 0x00, 0,0,0,0,0,0,0,0 };
        await using var stream = new MemoryStream(bytes);

        var result = await _validator.ValidateAsync(stream);

        result.IsValid.Should().BeFalse();
        result.FileType.Should().Be("Unknown");
    }

    [Fact]
    public async Task ValidateAsync_EmptyStream_ReturnsInvalid()
    {
        await using var stream = new MemoryStream(Array.Empty<byte>());

        var result = await _validator.ValidateAsync(stream);

        result.IsValid.Should().BeFalse();
    }

    [Fact]
    public async Task ValidateAsync_TruncatedShorterThanSignature_ReturnsInvalid()
    {
        // Only 2 bytes — shorter than any registered signature

        var result = await _validator.ValidateAsync(stream);

        result.IsValid.Should().BeFalse();
    }
}
```
