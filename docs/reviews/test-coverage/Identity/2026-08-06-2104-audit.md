#### HIGH: PrivacyErasureRequestedConsumer swallows the failure path silently
- **For**: `src/Identity/Identity.Application/Consumers/PrivacyErasureRequestedConsumer.cs:Consume`
- **Type**: Unit + Integration
- **Why**: GDPR erasure is a compliance-critical, one-shot operation. If `UserManager.UpdateAsync` fails, the consumer must publish `PrivacyErasureFailed` — this branch has never been exercised. A regression here means the Privacy saga hangs waiting for a completion event that never comes, with no observable failure signal.
- **Skeleton**:
```csharp

{

    private readonly Mock<IUserProfileRepository> _profilesMock;

    public PrivacyErasureRequestedConsumerTests(ITestOutputHelper output) : base(output)
    {
        var store = new Mock<IUserStore<User>>();
        _userManagerMock = new Mock<UserManager<User>>(store.Object, null!, null!, null!, null!, null!, null!, null!, null!);
        _profilesMock = new Mock<IUserProfileRepository>();
    }

    [Fact]
    public async Task Consume_WhenUpdateAsyncFails_PublishesPrivacyErasureFailed()
    {
        var userId = Guid.NewGuid();
        var user = new User { Id = userId.ToString(), UserName = "victim", Email = "victim@test.com" };
        _userManagerMock.Setup(m => m.FindByIdAsync(userId.ToString())).ReturnsAsync(user);
        _userManagerMock.Setup(m => m.UpdateAsync(It.IsAny<User>()))
            .ReturnsAsync(IdentityResult.Failed(new IdentityError { Description = "concurrency stamp mismatch" }));

        var consumeContextMock = new Mock<ConsumeContext<PrivacyErasureRequested>>();
        consumeContextMock.SetupGet(c => c.Message).Returns(new PrivacyErasureRequested
        {
            RequestId = Guid.NewGuid(), UserId = userId
        });
        consumeContextMock.SetupGet(c => c.CancellationToken).Returns(CancellationToken.None);

        PrivacyErasureFailed? published = null;
        consumeContextMock
            .Setup(c => c.Publish(It.IsAny<PrivacyErasureFailed>(), It.IsAny<CancellationToken>()))
            .Callback<object, CancellationToken>((msg, _) => published = (PrivacyErasureFailed)msg)
            .Returns(Task.CompletedTask);

        var consumer = new PrivacyErasureRequestedConsumer(
            _userManagerMock.Object, _profilesMock.Object, LoggerFactory.CreateLogger<PrivacyErasureRequestedConsumer>());

        await consumer.Consume(consumeContextMock.Object);

        published.Should().NotBeNull();
        published!.UserId.Should().Be(userId);
        consumeContextMock.Verify(c => c.Publish(It.IsAny<PrivacyErasureCompleted>(), It.IsAny<CancellationToken>()), Times.Never);

    public async Task Consume_WhenUserNotFound_StillAnonymizesProfileAndPublishesCompleted()

        var profile = UserProfile.Create(userId.ToString());
        _profilesMock.Setup(r => r.GetByUserIdAsync(userId.ToString(), It.IsAny<CancellationToken>())).ReturnsAsync(profile);

        var consumeContextMock = new Mock<ConsumeContext<PrivacyErasureRequested>>();
        consumeContextMock.SetupGet(c => c.Message).Returns(new PrivacyErasureRequested { RequestId = Guid.NewGuid(), UserId = userId });
        consumeContextMock.SetupGet(c => c.CancellationToken).Returns(CancellationToken.None);
        consumeContextMock.Setup(c => c.Publish(It.IsAny<PrivacyErasureCompleted>(), It.IsAny<CancellationToken>())).Returns(Task.CompletedTask);

        var consumer = new PrivacyErasureRequestedConsumer(
            _userManagerMock.Object, _profilesMock.Object, LoggerFactory.CreateLogger<PrivacyErasureRequestedConsumer>());

        await consumer.Consume(consumeContextMock.Object);

        profile.FirstName.Should().Be("DELETED");
        consumeContextMock.Verify(c => c.Publish(It.Is<PrivacyErasureCompleted>(e => e.UserId == userId), It.IsAny<CancellationToken>()), Times.Once);
    }
```
