#### HIGH: RotatingJwtSigningKeyRing rotation and grace-window expiry
- **For**: `RotatingJwtSigningKeyRing.cs:PollOnceAsync`, `SeedFromVaultAsync`, `BuildEntry`
- **Type**: Unit
- **Why**: This is the mechanism that keeps JWT validation working during key rotation. An undetected regression here (e.g. failing to demote the old key, or dropping a key before its grace window elapses) causes either a security hole (old key never expires) or an outage (in-flight tokens rejected mid-rotation). Zero tests exist.
- **Skeleton**:
```csharp
public class RotatingJwtSigningKeyRingTests
{
    private static IVaultService FakeVault(string pem) 
    {
        var mock = new Mock<IVaultService>();
        mock.Setup(v => v.GetKvSecretAsync(
                RotatingJwtSigningKeyRing.VaultKvPath,
                RotatingJwtSigningKeyRing.VaultKvField,
                It.IsAny<CancellationToken>()))
            .ReturnsAsync(pem);
        return mock.Object;
    }

    [Fact]
    public async Task PollOnceAsync_WhenVaultPemChanges_DemotesOldKeyToRetiring()
    {
        using var rsa1 = RSA.Create(2048);
        using var rsa2 = RSA.Create(2048);
        var pem1 = rsa1.ExportRSAPrivateKeyPem();
        var pem2 = rsa2.ExportRSAPrivateKeyPem();

        var vaultMock = new Mock<IVaultService>();
        vaultMock.SetupSequence(v => v.GetKvSecretAsync(
                RotatingJwtSigningKeyRing.VaultKvPath,
                RotatingJwtSigningKeyRing.VaultKvField,
                It.IsAny<CancellationToken>()))
            .ReturnsAsync(pem1)
            .ReturnsAsync(pem2);
        vaultMock.Setup(v => v.InitializeAsync(It.IsAny<CancellationToken>())).Returns(Task.CompletedTask);

        var timeProvider = new FakeTimeProvider(DateTimeOffset.UtcNow);
        var ring = new RotatingJwtSigningKeyRing(vaultMock.Object, NullLogger<RotatingJwtSigningKeyRing>.Instance, timeProvider);

        await ring.StartAsync(CancellationToken.None); // seeds Active with pem1
        var originalActiveKid = ring.Active.Kid;

        await ring.PollOnceAsync(CancellationToken.None); // detects pem2

        ring.Active.Kid.Should().NotBe(originalActiveKid);
        ring.AllValidKeys.Should().Contain(k => k.Kid == originalActiveKid && k.Status == JwtSigningKeyStatus.Retiring);
        ring.AllValidKeys.Should().HaveCount(2);
    }

    [Fact]
    public async Task PollOnceAsync_AfterGraceWindowElapses_DropsRetiringKey()
    {
        using var rsa1 = RSA.Create(2048);
        using var rsa2 = RSA.Create(2048);
        var pem1 = rsa1.ExportRSAPrivateKeyPem();
        var pem2 = rsa2.ExportRSAPrivateKeyPem();

        var vaultMock = new Mock<IVaultService>();
        vaultMock.Setup(v => v.InitializeAsync(It.IsAny<CancellationToken>())).Returns(Task.CompletedTask);
        vaultMock.SetupSequence(v => v.GetKvSecretAsync(
                RotatingJwtSigningKeyRing.VaultKvPath, RotatingJwtSigningKeyRing.VaultKvField, It.IsAny<CancellationToken>()))
            .ReturnsAsync(pem1)
            .ReturnsAsync(pem2)
            .ReturnsAsync(pem2);

        var start = DateTimeOffset.UtcNow;
        var timeProvider = new FakeTimeProvider(start);
        var ring = new RotatingJwtSigningKeyRing(vaultMock.Object, NullLogger<RotatingJwtSigningKeyRing>.Instance, timeProvider);

        await ring.StartAsync(CancellationToken.None);
        await ring.PollOnceAsync(CancellationToken.None); // rotate -> old key Retiring w/ 1h grace

        timeProvider.Advance(RotatingJwtSigningKeyRing.RetirementGrace + TimeSpan.FromMinutes(1));
        await ring.PollOnceAsync(CancellationToken.None);

        ring.AllValidKeys.Should().HaveCount(1);
        ring.AllValidKeys.Single().Status.Should().Be(JwtSigningKeyStatus.Active);
    }

    [Fact]
    public async Task SeedFromVaultAsync_WhenNoKeyInVault_GeneratesEphemeralKey()
    {
        var vault = FakeVault(pem: null!); // empty/whitespace
        var ring = new RotatingJwtSigningKeyRing(vault, NullLogger<RotatingJwtSigningKeyRing>.Instance);

        await ring.StartAsync(CancellationToken.None);

        ring.Active.Should().NotBeNull();
        ring.Active.Status.Should().Be(JwtSigningKeyStatus.Active);
    }

    [Fact]
    public async Task SeedFromVaultAsync_WhenVaultHasUnparseablePlaceholder_FallsBackToEphemeralKey()
    {
        var vault = FakeVault(pem: "dev-only-not-for-prod");
        var ring = new RotatingJwtSigningKeyRing(vault, NullLogger<RotatingJwtSigningKeyRing>.Instance);

        await ring.StartAsync(CancellationToken.None);

        ring.Active.Should().NotBeNull(); // must not throw / crash boot
    }
}
```
